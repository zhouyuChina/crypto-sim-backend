## REST API 接口

### 2.1 获取对话消息（用户端）

**请求**
```http
GET /api/support/messages?conversationId={conversationId}&limit=50&offset=0
Authorization: Bearer {user_token}
```

**响应**
```json
{
  "messages": [
    {
      "id": "msg-001",
      "conversationId": "conv-123",
      "senderId": "admin-456",
      "senderType": "ADMIN",
      "senderName": "客服小王",
      "messageType": "TEXT",
      "content": "您好，有什么可以帮您？",
      "metadata": null,
      "isRead": false,         // ✅ 已读状态
      "readAt": null,          // ✅ 已读时间
      "createdAt": "2025-01-14T10:00:00.000Z",
      "updatedAt": "2025-01-14T10:00:00.000Z"
    }
  ],
  "total": 10,
  "hasMore": false
}
```

**注意**：消息数组中**自动包含** `isRead` 和 `readAt` 字段，无需额外请求。

### 2.2 标记消息已读（用户端）

**请求**
```http
PUT /api/support/messages/read
Authorization: Bearer {user_token}
Content-Type: application/json

{
  "conversationId": "conv-123"
}
```

**响应**
```json
{
  "message": "已标记为已读"
}
```

**效果**：
1. 将该对话中所有 `senderType='ADMIN'` 且 `isRead=false` 的消息标记为已读
2. 将 `ChatConversation.unreadUserCount` 清零
3. 通过 WebSocket 向对话房间推送 `support:messages-read` 事件

## 完整集成示例

### 4.1 用户端 - React 示例

```tsx
import React, { useEffect, useState, useRef } from 'react';
import { io, Socket } from 'socket.io-client';
import axios from 'axios';

interface ChatMessage {
  id: string;
  conversationId: string;
  senderId: string;
  senderType: 'USER' | 'ADMIN' | 'SYSTEM';
  senderName: string | null;
  messageType: 'TEXT' | 'IMAGE' | 'SYSTEM';
  content: string;
  isRead: boolean;
  readAt: string | null;
  createdAt: string;
}

export const UserChatComponent: React.FC = () => {
  const [messages, setMessages] = useState<ChatMessage[]>([]);
  const [conversationId, setConversationId] = useState<string>('');
  const socketRef = useRef<Socket | null>(null);

  // 1. 初始化 WebSocket 连接
  useEffect(() => {
    const token = localStorage.getItem('userToken');

    socketRef.current = io('http://localhost:3000/support', {
      auth: { token }
    });

    // 监听已读通知
    socketRef.current.on('support:messages-read', (data) => {
      console.log('对方已读', data);

      if (data.readerType === 'admin') {
        // 管理员已读我的消息，更新消息状态
        setMessages(prevMessages =>
          prevMessages.map(msg =>
            msg.senderType === 'USER' && !msg.isRead
              ? { ...msg, isRead: true, readAt: data.readAt }
              : msg
          )
        );
      }
    });

    // 监听新消息
    socketRef.current.on('support:message', (message: ChatMessage) => {
      setMessages(prev => [...prev, message]);

      // 如果是对方发来的消息，自动标记已读（可选）
      if (message.senderType === 'ADMIN') {
        markAsReadDebounced(message.conversationId);
      }
    });

    return () => {
      socketRef.current?.disconnect();
    };
  }, []);

  // 2. 获取或创建对话
  useEffect(() => {
    const initConversation = async () => {
      const token = localStorage.getItem('userToken');
      const response = await axios.get('/api/support/conversation', {
        headers: { Authorization: `Bearer ${token}` }
      });

      const conv = response.data;
      setConversationId(conv.id);
      setMessages(conv.messages || []);

      // 加入 WebSocket 房间
      socketRef.current?.emit('support:join', {
        conversationId: conv.id,
        userType: 'user'
      });

      // 进入对话时标记已读
      markAsRead(conv.id);
    };

    initConversation();
  }, []);

  // 3. 标记消息已读（REST API）
  const markAsRead = async (convId: string) => {
    try {
      const token = localStorage.getItem('userToken');
      await axios.put('/api/support/messages/read',
        { conversationId: convId },
        { headers: { Authorization: `Bearer ${token}` }}
      );
      console.log('已标记为已读');
    } catch (error) {
      console.error('标记已读失败', error);
    }
  };

  // 4. 防抖标记已读（避免频繁调用）
  const markAsReadDebounced = debounce(markAsRead, 1000);

  // 5. 发送消息
  const sendMessage = (content: string) => {
    socketRef.current?.emit('support:message', {
      conversationId,
      senderId: 'current-user-id',
      senderType: 'USER',
      senderName: '用户名',
      messageType: 'TEXT',
      content
    });
  };

  // 6. 渲染消息列表
  return (
    <div className="chat-container">
      <div className="message-list">
        {messages.map(msg => (
          <div key={msg.id} className={`message ${msg.senderType}`}>
            <div className="message-content">{msg.content}</div>
            <div className="message-meta">
              <span>{new Date(msg.createdAt).toLocaleTimeString()}</span>

              {/* ✅ 显示已读状态 */}
              {msg.senderType === 'USER' && (
                <span className="read-status">
                  {msg.isRead ? (
                    <span className="read">✓✓ 已读</span>
                  ) : (
                    <span className="unread">✓ 未读</span>
                  )}
                </span>
              )}
            </div>
          </div>
        ))}
      </div>
    </div>
  );
};

// 防抖函数
function debounce<T extends (...args: any[]) => any>(
  func: T,
  wait: number
): (...args: Parameters<T>) => void {
  let timeout: NodeJS.Timeout;
  return (...args: Parameters<T>) => {
    clearTimeout(timeout);
    timeout = setTimeout(() => func(...args), wait);
  };
}
```

### 4.2 管理员端 - React 示例

```tsx
import React, { useEffect, useState, useRef } from 'react';
import { io, Socket } from 'socket.io-client';
import axios from 'axios';

export const AdminChatComponent: React.FC<{ conversationId: string }> = ({
  conversationId
}) => {
  const [messages, setMessages] = useState<ChatMessage[]>([]);
  const [unreadCount, setUnreadCount] = useState<number>(0);
  const socketRef = useRef<Socket | null>(null);

  useEffect(() => {
    const token = localStorage.getItem('adminToken');

    // 1. 连接 WebSocket
    socketRef.current = io('http://localhost:3000/support', {
      auth: { token }
    });

    // 2. 监听已读通知
    socketRef.current.on('support:messages-read', (data) => {
      if (data.conversationId === conversationId) {
        if (data.readerType === 'user') {
          // 用户已读管理员的消息
          setMessages(prevMessages =>
            prevMessages.map(msg =>
              msg.senderType === 'ADMIN' && !msg.isRead
                ? { ...msg, isRead: true, readAt: data.readAt }
                : msg
            )
          );
        }
      }
    });

    // 3. 监听新消息
    socketRef.current.on('support:message', (message: ChatMessage) => {
      if (message.conversationId === conversationId) {
        setMessages(prev => [...prev, message]);

        // 如果是用户发来的消息
        if (message.senderType === 'USER') {
          setUnreadCount(prev => prev + 1);
        }
      }
    });

    // 4. 加入房间
    socketRef.current.emit('support:join', {
      conversationId,
      userType: 'admin'
    });

    // 5. 加载消息历史
    loadMessages();

    return () => {
      socketRef.current?.disconnect();
    };
  }, [conversationId]);

  // 加载消息
  const loadMessages = async () => {
    try {
      const token = localStorage.getItem('adminToken');
      const response = await axios.get(
        `/api/admin/support/messages?conversationId=${conversationId}`,
        { headers: { Authorization: `Bearer ${token}` }}
      );

      const msgs = response.data.messages;
      setMessages(msgs);

      // 计算未读数
      const unread = msgs.filter(
        (m: ChatMessage) => m.senderType === 'USER' && !m.isRead
      ).length;
      setUnreadCount(unread);
    } catch (error) {
      console.error('加载消息失败', error);
    }
  };

  // 标记已读（管理员端）
  const markAsRead = () => {
    socketRef.current?.emit('support:read', {
      conversationId,
      readerType: 'admin'
    });

    // 立即更新本地状态
    setUnreadCount(0);
    setMessages(prevMessages =>
      prevMessages.map(msg =>
        msg.senderType === 'USER' && !msg.isRead
          ? { ...msg, isRead: true, readAt: new Date().toISOString() }
          : msg
      )
    );
  };

  // 渲染
  return (
    <div className="admin-chat">
      <div className="chat-header">
        <h3>对话详情</h3>
        {unreadCount > 0 && (
          <button onClick={markAsRead}>
            标记所有消息已读 ({unreadCount})
          </button>
        )}
      </div>

      <div className="message-list">
        {messages.map(msg => (
          <div key={msg.id} className={`message ${msg.senderType}`}>
            <div className="message-content">{msg.content}</div>
            <div className="message-meta">
              <span>{new Date(msg.createdAt).toLocaleTimeString()}</span>

              {/* ✅ 管理员发送的消息显示已读状态 */}
              {msg.senderType === 'ADMIN' && (
                <span className={msg.isRead ? 'read' : 'unread'}>
                  {msg.isRead ? '✓✓ 已读' : '✓ 未读'}
                  {msg.readAt && ` (${new Date(msg.readAt).toLocaleTimeString()})`}
                </span>
              )}
            </div>
          </div>
        ))}
      </div>
    </div>
  );
};
```

### 5.2 防止重复标记

```typescript
let lastMarkTime = 0;
const MARK_INTERVAL = 5000; // 5秒内不重复标记

const markAsReadThrottled = (conversationId: string) => {
  const now = Date.now();
  if (now - lastMarkTime < MARK_INTERVAL) {
    return; // 跳过
  }

  lastMarkTime = now;
  markAsRead(conversationId);
};
```

## TypeScript 类型定义

```typescript
// types/chat.ts

export enum SenderType {
  USER = 'USER',
  ADMIN = 'ADMIN',
  SYSTEM = 'SYSTEM'
}

export enum MessageType {
  TEXT = 'TEXT',
  IMAGE = 'IMAGE',
  SYSTEM = 'SYSTEM'
}

export enum ConversationStatus {
  PENDING = 'PENDING',
  ACTIVE = 'ACTIVE',
  CLOSED = 'CLOSED'
}

export interface ChatMessage {
  id: string;
  conversationId: string;
  senderId: string;
  senderType: SenderType;
  senderName: string | null;
  messageType: MessageType;
  content: string;
  metadata: any | null;
  isRead: boolean;
  readAt: string | null;
  createdAt: string;
  updatedAt: string;
}

export interface ChatConversation {
  id: string;
  userId: string;
  userName: string | null;
  adminId: string | null;
  adminName: string | null;
  status: ConversationStatus;
  unreadUserCount: number;
  unreadAdminCount: number;
  lastMessageAt: string | null;
  lastMessage: string | null;
  lastMessageType: MessageType | null;
  createdAt: string;
  updatedAt: string;
  closedAt: string | null;
  messages?: ChatMessage[];
}

// WebSocket 事件类型
export interface SocketEvents {
  // 客户端发送
  'support:join': {
    conversationId: string;
    userType: 'user' | 'admin';
  };

  'support:message': {
    conversationId: string;
    senderId: string;
    senderType: SenderType;
    senderName: string;
    messageType: MessageType;
    content: string;
    metadata?: any;
  };

  'support:read': {
    conversationId: string;
    readerType: 'user' | 'admin';
  };

  'support:leave': {
    conversationId: string;
  };

  // 服务器推送
  'support:joined': {
    conversationId: string;
    roomName: string;
  };

  'support:messages-read': {
    conversationId: string;
    readerType: 'user' | 'admin';
    readAt: string;
  };
}
```

## 常见问题 FAQ

### Q1: 获取消息时是否需要手动请求 isRead 字段？
**A:** 不需要。后端 `getMessages()` 方法使用 Prisma 默认返回所有字段，包括 `isRead` 和 `readAt`。

### Q2: 标记已读后，对方多久能收到通知？
**A:** 实时推送，延迟通常在 50-200ms 之间（取决于网络状况）。

### Q3: 如果用户离线，消息会自动标记已读吗？
**A:** 不会。只有调用 `markAsRead` 接口或发送 WebSocket 事件才会标记。

### Q4: 一次标记已读会标记多少条消息？
**A:** 标记该对话中所有**对方发送的未读消息**。用户调用时标记所有管理员消息，管理员调用时标记所有用户消息。

### Q5: 如何实现单条消息的已读回执？
**A:** 后端已支持。前端可以：
1. 监听 `support:messages-read` 事件获取 `readAt` 时间
2. 更新所有符合条件的消息状态（`isRead=true`）
3. UI 上显示已读时间

### Q6: 支持多设备同步吗？
**A:** 支持。同一用户在多个设备登录，标记已读后会通过 WebSocket 推送到所有设备。
